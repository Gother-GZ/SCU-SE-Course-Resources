## 勘误 1：4.2.2链式栈（PDF 第 17 页）
```cpp
template <typename T>
class Link {
	public:
        T element;
        Link *next;
        
        Link(const T& eleVal, Link* nextVal = NULL) {
            element = eleVal;
            next = nextVal;
        }
};

template <typename T>
class LStack {
    private:
        Link<T>* top;
        int size;

    public:
        LStack() {
            top = NULL;
            size = 0;
        }

        ~LStack() {
            clear();
        }

        void clear() {
			while (top != NULL) {
                Link<T>* temp = top;
                top = top->next;
                delete temp;
            }
            size = 0;
        }

        void push(const T& it) {
            Link<T>* temp = new Link<T>(it);
            temp->next = top;
            top = temp;     //不要忘了更新栈顶
            size++;
        }

        T pop() {
            static_assert(top != NULL, "Stack is empty.");
            Link<T>* temp = top;
            top = top->next;
            size--;
            T it = temp->element;
            delete temp;
            return it;
        }

        T getVal() const {
            static_assert(top != NULL, "Stack is empty.");
            return top->element;
        }

        int length()const {
            return size;
        }

};
```
